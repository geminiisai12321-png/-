        function spawnPracticeEnemy(type) {

        class Enemy {
            constructor(type, waveBonus) {
                this.x = path[0].x;
                this.y = path[0].y;
                this.pathIndex = 0;
                this.slowTimer = 0;

                const healthMultiplier = Math.pow(1.08, Math.max(0, waveBonus - 10));

                if (type === 'fast') {
                    this.speed = 2.2;
                    this.maxHp = Math.floor((60 + waveBonus * 25) * healthMultiplier);
                    this.color = '#e11d48';
                    this.radius = 10;
                    this.reward = 15;
                } else if (type === 'boss') {
                    this.speed = 0.8;
                    this.maxHp = Math.floor((500 + waveBonus * 200) * healthMultiplier);
                    this.color = '#7e22ce';
                    this.radius = 20;
                    this.reward = 100;
                } else if (type === 'armored') {
                    this.speed = 1.1;
                    this.maxHp = Math.floor((180 + waveBonus * 65) * healthMultiplier);
                    this.color = '#475569';
                    this.radius = 15;
                    this.reward = 35;
                } else {
                    this.speed = 1.4;
                    this.maxHp = Math.floor((90 + waveBonus * 35) * healthMultiplier);
                    this.color = '#ea580c';
                    this.radius = 12;
                    this.reward = 20;
                }
                this.hp = this.maxHp;
            }

            update() {
                if (this.slowTimer > 0) this.slowTimer--;

                const currentSpeed = (this.slowTimer > 0) ? this.speed * 0.5 : this.speed;
                const target = path[this.pathIndex + 1];

                if (!target) return;

                const dx = target.x - this.x;
                const dy = target.y - this.y;
                const dist = Math.hypot(dx, dy);

                if (dist < currentSpeed * gameSpeed) {
                    this.x = target.x;
                    this.y = target.y;
                    this.pathIndex++;
                    if (this.pathIndex >= path.length - 1) {
                        if (currentMode !== 'practice') {
                            lives--;
                            livesDisplay.textContent = lives;
                        }
                        this.hp = 0;
                        if (lives <= 0 && currentMode !== 'practice') handleGameOver(false);
                    }
                } else {
                    this.x += (dx / dist) * currentSpeed * gameSpeed;
                    this.y += (dy / dist) * currentSpeed * gameSpeed;
                }
            }

            draw() {
                ctx.save();
                ctx.translate(this.x, this.y);

                ctx.fillStyle = this.slowTimer > 0 ? '#06b6d4' : this.color;
                ctx.beginPath();
                ctx.arc(0, 0, this.radius, 0, Math.PI * 2);
                ctx.fill();
                ctx.strokeStyle = '#ffffff';
                ctx.lineWidth = 1.5;
                ctx.stroke();

                // HP Bar
                const barW = this.radius * 2.2;
                const barH = 4;
                ctx.fillStyle = 'rgba(0, 0, 0, 0.6)';
                ctx.fillRect(-barW / 2, -this.radius - 8, barW, barH);

                const hpRatio = Math.max(0, this.hp / this.maxHp);
                ctx.fillStyle = hpRatio > 0.5 ? '#22c55e' : hpRatio > 0.2 ? '#eab308' : '#ef4444';
                ctx.fillRect(-barW / 2, -this.radius - 8, barW * hpRatio, barH);

                ctx.restore();
            }
        }

        class Tower {
            constructor(x, y, type) {
                this.x = x;
                this.y = y;
                this.type = type;
                this.level = 1;
                this.config = { ...TOWER_TYPES[type] };
                this.cooldown = 0;
            }

            getUpgradeCost() {
                return Math.floor(this.config.cost * Math.pow(1.5, this.level));
            }

            update() {
                if (this.cooldown > 0) this.cooldown -= gameSpeed;

                if (this.cooldown <= 0) {
                    let target = null;
                    let minDist = this.config.range;

                    for (let enemy of enemies) {
                        const dist = Math.hypot(enemy.x - this.x, enemy.y - this.y);
                        if (dist <= minDist) {
                            minDist = dist;
                            target = enemy;
                        }
                    }

                    if (target) {
                        this.fire(target);
                        this.cooldown = this.config.fireRate;
                    }
                }
            }

            fire(target) {
                const bulletCount = this.level;
                const baseAngle = Math.atan2(target.y - this.y, target.x - this.x);
                const spreadAngle = 0.15;

                const startAngle = baseAngle - ((bulletCount - 1) * spreadAngle) / 2;

                for (let i = 0; i < bulletCount; i++) {
                    const angle = startAngle + i * spreadAngle;
                    projectiles.push(new Projectile(this, target, angle));
                }
            }

            draw() {
                ctx.save();
                ctx.translate(this.x, this.y);

                // Base platform
                ctx.fillStyle = '#334155';
                ctx.beginPath();
                ctx.arc(0, 0, TOWER_RADIUS + 2, 0, Math.PI * 2);
                ctx.fill();

                // Inner Body
                ctx.fillStyle = this.config.color;
                ctx.beginPath();
                ctx.arc(0, 0, TOWER_RADIUS - 4, 0, Math.PI * 2);
                ctx.fill();

                // Display Stars for Level
                ctx.fillStyle = '#fde047';
                const startX = -((this.level - 1) * 6) / 2;
                for (let i = 0; i < this.level; i++) {
                    ctx.beginPath();
                    ctx.arc(startX + i * 6, 12, 2.5, 0, Math.PI * 2);
                    ctx.fill();
                }

                ctx.restore();

                // Range Indicator
                if (selectedTower === this) {
                    ctx.save();
                    ctx.beginPath();
                    ctx.arc(this.x, this.y, this.config.range, 0, Math.PI * 2);
                    ctx.fillStyle = 'rgba(59, 130, 246, 0.15)';
                    ctx.fill();
                    ctx.strokeStyle = '#3b82f6';
                    ctx.lineWidth = 1.5;
                    ctx.stroke();
                    ctx.restore();
                }
            }

            upgrade() {
                if (this.level >= 5) return;

                const cost = this.getUpgradeCost();
                if (gold >= cost || currentMode === 'practice') {
                    if (currentMode !== 'practice') {
                        gold -= cost;
                        goldDisplay.textContent = gold;
                    }
                    this.level++;
                    this.config.damage = Math.floor(this.config.damage * 1.35);
                    this.config.range = Math.floor(this.config.range * 1.08);
                    updateInfoPanel();
                }
            }

            sell() {
                if (currentMode !== 'practice') {
                    let totalInvested = this.config.cost;
                    for (let lvl = 1; lvl < this.level; lvl++) {
                        totalInvested += Math.floor(this.config.cost * Math.pow(1.5, lvl));
                    }
                    const returnGold = Math.floor(totalInvested * 0.6);
                    gold += returnGold;
                    goldDisplay.textContent = gold;
                }
                towers = towers.filter(t => t !== this);
                deselectTower();
            }
        }

        class Projectile {
            constructor(tower, target, angle) {
                this.x = tower.x;
                this.y = tower.y;
                this.tower = tower;
                this.target = target;
                this.speed = tower.config.bulletSpeed;
                this.vx = Math.cos(angle) * this.speed;
                this.vy = Math.sin(angle) * this.speed;
                this.life = 120;
            }

            update() {
                this.x += this.vx * gameSpeed;
                this.y += this.vy * gameSpeed;
                this.life -= gameSpeed;

                for (let enemy of enemies) {
                    if (Math.hypot(enemy.x - this.x, enemy.y - this.y) < enemy.radius + 4) {
                        this.hit(enemy);
                        return true;
                    }
                }

                return this.life <= 0 || this.x < 0 || this.x > BASE_WIDTH || this.y < 0 || this.y > BASE_HEIGHT;
            }

            hit(enemy) {
                const cfg = this.tower.config;
                if (cfg.splash) {
                    enemies.forEach(e => {
                        if (Math.hypot(e.x - this.x, e.y - this.y) <= cfg.splash) {
                            e.hp -= cfg.damage;
                            if (cfg.slow) e.slowTimer = 90;
                        }
                    });
                    createExplosion(this.x, this.y, cfg.color);
                } else {
                    enemy.hp -= cfg.damage;
                    if (cfg.slow) enemy.slowTimer = 120;
                    createExplosion(this.x, this.y, cfg.color);
                }

                if (enemy.hp <= 0 && enemy.rewardGiven !== true) {
                    enemy.rewardGiven = true;
                    if (currentMode !== 'practice') {
                        gold += enemy.reward;
                        goldDisplay.textContent = gold;
                    }
                }
            }

            draw() {
                ctx.save();
                ctx.fillStyle = this.tower.config.color;
                ctx.beginPath();
                ctx.arc(this.x, this.y, 4, 0, Math.PI * 2);
                ctx.fill();
                ctx.restore();
            }
        }

        function isPointOnPath(x, y) {
            for (let i = 0; i < path.length - 1; i++) {
                const p1 = path[i];
                const p2 = path[i + 1];

                const l2 = Math.hypot(p2.x - p1.x, p2.y - p1.y);
                if (l2 === 0) continue;

                let t = ((x - p1.x) * (p2.x - p1.x) + (y - p1.y) * (p2.y - p1.y)) / (l2 * l2);
                t = Math.max(0, Math.min(1, t));

                const projX = p1.x + t * (p2.x - p1.x);
                const projY = p1.y + t * (p2.y - p1.y);

                if (Math.hypot(x - projX, y - projY) < PATH_WIDTH / 2 + TOWER_RADIUS) {
                    return true;
                }
            }
            return false;
        }

        function isValidBuildPosition(x, y) {
            if (isPointOnPath(x, y)) return false;
            for (let t of towers) {
                if (Math.hypot(t.x - x, t.y - y) < TOWER_RADIUS * 2) return false;
            }
            return true;
        }

        function drawTree(x, y) {
            ctx.fillStyle = '#15803d';
            ctx.beginPath(); ctx.arc(x, y, 16, 0, Math.PI * 2); ctx.fill();
            ctx.fillStyle = '#166534';
            ctx.beginPath(); ctx.arc(x - 5, y - 5, 10, 0, Math.PI * 2); ctx.fill();
        }

        function drawDecorations() {
            currentMap.decorations.forEach(dec => {
                if (dec.type === 'tree') {
                    drawTree(dec.x, dec.y);
                } else if (dec.type === 'rock') {
                    ctx.fillStyle = '#64748b';
                    ctx.beginPath(); ctx.arc(dec.x, dec.y, 14, 0, Math.PI * 2); ctx.fill();
                    ctx.fillStyle = '#475569';
                    ctx.beginPath(); ctx.arc(dec.x - 3, dec.y - 3, 8, 0, Math.PI * 2); ctx.fill();
                } else if (dec.type === 'crystal') {
                    ctx.fillStyle = '#38bdf8';
                    ctx.beginPath();
                    ctx.moveTo(dec.x, dec.y - 15);
                    ctx.lineTo(dec.x + 10, dec.y + 10);
                    ctx.lineTo(dec.x - 10, dec.y + 10);
                    ctx.closePath();
                    ctx.fill();
                } else if (dec.type === 'lava') {
                    ctx.fillStyle = '#ef4444';
                    ctx.beginPath(); ctx.arc(dec.x, dec.y, 18, 0, Math.PI * 2); ctx.fill();
                    ctx.fillStyle = '#f97316';
                    ctx.beginPath(); ctx.arc(dec.x, dec.y, 10, 0, Math.PI * 2); ctx.fill();
                }
            });
        }

        function drawBackground() {
            ctx.fillStyle = currentMap.bgColor;
            ctx.fillRect(0, 0, BASE_WIDTH, BASE_HEIGHT);

            ctx.strokeStyle = currentMap.pathOuterColor;
            ctx.lineWidth = PATH_WIDTH;
            ctx.lineCap = 'round';
            ctx.lineJoin = 'round';
            ctx.beginPath();
            ctx.moveTo(path[0].x, path[0].y);
            for (let i = 1; i < path.length; i++) {
                ctx.lineTo(path[i].x, path[i].y);
            }
            ctx.stroke();

            ctx.strokeStyle = currentMap.pathInnerColor;
            ctx.lineWidth = PATH_WIDTH - 8;
            ctx.stroke();

            drawDecorations();

            if (selectedTowerType && mouseX > 0 && mouseY > 0) {
                const valid = isValidBuildPosition(mouseX, mouseY);
                const cfg = TOWER_TYPES[selectedTowerType];

                ctx.save();
                ctx.beginPath();
                ctx.arc(mouseX, mouseY, cfg.range, 0, Math.PI * 2);
                ctx.fillStyle = valid ? 'rgba(34, 197, 94, 0.15)' : 'rgba(239, 68, 68, 0.15)';
                ctx.fill();
                ctx.strokeStyle = valid ? '#22c55e' : '#ef4444';
                ctx.lineWidth = 1.5;
                ctx.stroke();

                ctx.beginPath();
                ctx.arc(mouseX, mouseY, TOWER_RADIUS, 0, Math.PI * 2);
                ctx.fillStyle = valid ? cfg.color : '#ef4444';
                ctx.globalAlpha = 0.6;
                ctx.fill();
                ctx.restore();
            }
        }

        function createExplosion(x, y, color) {
            for (let i = 0; i < 5; i++) {
                particles.push({
                    x, y,
                    vx: (Math.random() - 0.5) * 4,
                    vy: (Math.random() - 0.5) * 4,
                    life: 12,
                    color
                });
            }
        }

        function spawnWave() {
            if (isWaveActive || isSpawning || gameOver || gameWon) return;

            if (spawnInterval) {
                clearInterval(spawnInterval);
                spawnInterval = null;
            }

            autoWaveCountdown = 0;
            currentWave++;
            if (currentMode === 'classic' && currentWave > maxWaves) return;

            waveDisplay.textContent = maxWaves === Infinity ? `${currentWave} / ∞` : `${currentWave} / ${maxWaves}`;
            isWaveActive = true;
            isSpawning = true;
            startWaveBtn.disabled = true;

            const count = 5 + currentWave * 3;
            let spawned = 0;

            spawnInterval = setInterval(() => {
                if (gameOver || gameWon) {
                    clearInterval(spawnInterval);
                    spawnInterval = null;
                    isSpawning = false;
                    return;
                }

                let type = 'normal';
                if (currentWave >= 3 && Math.random() < 0.35) type = 'fast';
                if (currentWave >= 5 && Math.random() < 0.3) type = 'armored';
                if (currentWave % 5 === 0 && spawned === count - 1) type = 'boss';

                enemies.push(new Enemy(type, currentWave));
                spawned++;

                if (spawned >= count) {
                    clearInterval(spawnInterval);
                    spawnInterval = null;
                    isSpawning = false;
                }
            }, 1000 / gameSpeed);
        }

        function gameLoop() {
            ctx.clearRect(0, 0, BASE_WIDTH, BASE_HEIGHT);

            drawBackground();

            towers.forEach(tower => {
                tower.update();
                tower.draw();
            });

            for (let i = enemies.length - 1; i >= 0; i--) {
                enemies[i].update();
                enemies[i].draw();
                if (enemies[i].hp <= 0) {
                    enemies.splice(i, 1);
                }
            }

            for (let i = projectiles.length - 1; i >= 0; i--) {
                if (projectiles[i].update()) {
                    projectiles.splice(i, 1);
                } else {
                    projectiles[i].draw();
                }
            }

            for (let i = particles.length - 1; i >= 0; i--) {
                const p = particles[i];
                p.x += p.vx;
                p.y += p.vy;
                p.life--;
                ctx.fillStyle = p.color;
                ctx.fillRect(p.x, p.y, 3, 3);
                if (p.life <= 0) particles.splice(i, 1);
            }

            if (isWaveActive && !isSpawning && enemies.length === 0) {
                isWaveActive = false;
                startWaveBtn.disabled = false;
                
                if (currentMode === 'classic' && currentWave >= maxWaves) {
                    handleGameOver(true);
                } else if (autoWave && !gameOver && !gameWon) {
                    autoWaveCountdown = Math.floor(70 / gameSpeed);
                }
            }

            if (!isWaveActive && !isSpawning && autoWave && enemies.length === 0 && !gameOver && !gameWon) {
                if (autoWaveCountdown > 0) {
                    autoWaveCountdown -= gameSpeed;
                    if (autoWaveCountdown <= 0) {
                        spawnWave();
                    }
                }
            }

            if (!gameOver && !gameWon) {
                requestAnimationFrame(gameLoop);
            }
        }

        function resetGame() {
            if (spawnInterval) {
                clearInterval(spawnInterval);
                spawnInterval = null;
            }

            autoWaveCountdown = 0;
            isSpawning = false;

            if (currentMode === 'practice') {
                gold = 99999;
                lives = 999;
            } else {
                gold = 250;
                lives = 20;
            }

            currentWave = 0;
            gameSpeed = 1;
            selectedTowerType = null;
            selectedTower = null;
            isWaveActive = false;
            gameOver = false;
            gameWon = false;

            towers = [];
            enemies = [];
            projectiles = [];
            particles = [];

            goldDisplay.textContent = currentMode === 'practice' ? '∞' : gold;
            livesDisplay.textContent = currentMode === 'practice' ? '∞' : lives;
            waveDisplay.textContent = maxWaves === Infinity ? `0 / ∞` : `0 / ${maxWaves}`;
            speedText.textContent = "1x";
            startWaveBtn.disabled = false;

            towerButtons.forEach(btn => btn.classList.remove('selected'));
            infoPanel.classList.add('hidden');
            modal.classList.add('hidden');

            requestAnimationFrame(gameLoop);
        }

        function handleGameOver(win) {
            modal.classList.remove('hidden');
            const title = document.getElementById('modal-title');
            const desc = document.getElementById('modal-desc');

            if (win) {
                gameWon = true;
                title.textContent = "王國勝利！";
                title.className = "text-3xl font-extrabold text-emerald-400 mb-2";
                desc.textContent = "您成功擊退了所有怪物波次，守護了王國和平！";
            } else {
                gameOver = true;
                title.textContent = currentMode === 'endless' ? "挑戰結束！" : "城堡陷落！";
                title.className = "text-3xl font-extrabold text-rose-500 mb-2";
                desc.textContent = currentMode === 'endless' 
                    ? `您在無限模式中成功生存了 ${currentWave} 波怪物！` 
                    : "防線被突破，請調整策略重新再來！";
            }
        }

        // Handle Mouse & Mobile Touch Coordinates Scaling
        function getCanvasCoordinates(e) {
            const rect = canvas.getBoundingClientRect();
            const clientX = e.touches ? e.touches[0].clientX : e.clientX;
            const clientY = e.touches ? e.touches[0].clientY : e.clientY;
            
            const scaleX = BASE_WIDTH / rect.width;
            const scaleY = BASE_HEIGHT / rect.height;

            return {
                x: (clientX - rect.left) * scaleX,
                y: (clientY - rect.top) * scaleY
            };
        }

        function handlePointerMove(e) {
            if (e.type === 'touchmove') e.preventDefault();
            const coords = getCanvasCoordinates(e);
            mouseX = coords.x;
            mouseY = coords.y;
        }

        function handlePointerLeave() {
            mouseX = -100;
            mouseY = -100;
        }

        function handlePointerClick(e) {
            if (e.type === 'touchend') e.preventDefault();
            const coords = getCanvasCoordinates(e.type === 'touchend' ? e.changedTouches[0] : e);
            const clickX = coords.x;
            const clickY = coords.y;

            const clickedTower = towers.find(t => Math.hypot(t.x - clickX, t.y - clickY) < TOWER_RADIUS + 5);
            if (clickedTower) {
                selectTowerInstance(clickedTower);
                return;
            }

            if (selectedTowerType) {
                if (isValidBuildPosition(clickX, clickY)) {
                    const config = TOWER_TYPES[selectedTowerType];
                    if (gold >= config.cost || currentMode === 'practice') {
                        if (currentMode !== 'practice') {
                            gold -= config.cost;
                            goldDisplay.textContent = gold;
                        }
                        towers.push(new Tower(clickX, clickY, selectedTowerType));
                        deselectTower();
                    }
                }
            } else {
                deselectTower();
            }
        }

        // Pointer / Touch Listeners for Desktop and Mobile/iPhone
        canvas.addEventListener('mousemove', handlePointerMove);
        canvas.addEventListener('touchmove', handlePointerMove, { passive: false });
        canvas.addEventListener('mouseleave', handlePointerLeave);
        canvas.addEventListener('click', handlePointerClick);
        canvas.addEventListener('touchend', handlePointerClick, { passive: false });

        towerButtons.forEach(btn => {
            btn.addEventListener('click', () => {
                deselectTower();
                const type = btn.dataset.type;
                if (selectedTowerType === type) {
                    selectedTowerType = null;
                    btn.classList.remove('selected');
                } else {
                    towerButtons.forEach(b => b.classList.remove('selected'));
                    selectedTowerType = type;
                    btn.classList.add('selected');
                }
            });
        });

            infoPanel.classList.remove('hidden');
        }

            document.getElementById('info-bullets').textContent = `${selectedTower.level} 發`;
            document.getElementById('info-range').textContent = selectedTower.config.range;

            const upgradeBtn = document.getElementById('btn-upgrade');
            const upgradeText = document.getElementById('upgrade-text');

            if (selectedTower.level >= 5) {
                upgradeBtn.disabled = true;
                upgradeText.textContent = "已達最高等級 (MAX)";
            } else {
                upgradeBtn.disabled = false;
                const cost = selectedTower.getUpgradeCost();
                upgradeText.innerHTML = `升級 (<span id="upgrade-cost">${cost}</span>)`;
            }

            let totalInvested = selectedTower.config.cost;
            for (let lvl = 1; lvl < selectedTower.level; lvl++) {
                totalInvested += Math.floor(selectedTower.config.cost * Math.pow(1.5, lvl));
            }
            document.getElementById('sell-value').textContent = Math.floor(totalInvested * 0.6);
        }
